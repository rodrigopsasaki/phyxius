export interface RetryLedger {
  remaining(): number;
  draw(want: number): number;
}

export function createRetryLedger(totalExtraAttempts: number): RetryLedger {
  let remaining = totalExtraAttempts;

  return {
    remaining: () => remaining,
    draw(want: number): number {
      if (want <= 0) return 0;
      const granted = Math.min(want, remaining);
      remaining -= granted;
      return granted;
    },
  };
}
